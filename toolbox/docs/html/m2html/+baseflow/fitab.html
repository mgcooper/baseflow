PURPOSE

FITAB fit event-scale recession equation -dq/dt = aQ^b

SYNOPSIS

function [Fit,ok] = fitab(q,dqdt,method,varargin)

DESCRIPTION 

FITAB fit event-scale recession equation -dq/dt = aQ^b

 Syntax

     [Fit,ok] = fitab(q,dqdt,method,varargin)

 Description

     [Fit,ok] = fitab(q,dqdt,method) fits event-scale recession equation
     -dq/dt = aQ^b to estimate parameters a and b using the specified
     fitting method. Valid methods are ordinary least squares (ols),
     non-linear least squares (nls), quantile regression (qtl), mean
     difference ('mean'), median difference ('median').

 Required inputs

     q        vector double of discharge data (L T^-1)
     dqdt     vector double of discharge rate of change (L T^-2)
     method   char indicating the fitting method

 Optional inputs

     weights  vector double of weights for the fitting algorithm
     mask     vector logical mask to exclude values from fitting
     order    scalar, exponent in -dq/dt = aQ^b
     refqtls  2x1 double, x/y quantiles used if 'method' == 'envelope'
     quantile scalar double, quantile used if 'method' == 'qtl' (quantile
              regression)
     Nboot    scalar double, bootstrap sample size for quantile regression
     plotfit  logical scalar indicating whether to make a plot or not
     fitopts  struct whose fields override the same-named options above
              (weights, order, mask, quantile, refqtls, Nboot, alpha,
              plotfit); an unknown field is an error

 Notes
     weights are set zero anywhere mask is false

     Use method 'envelope' to pass a line through an arbitrary x,y pair
     (refpoints), specified in terms of the quantile of the x/y data
     distributions. this method is similar to 'mean' or 'median', but allows
     different ref points for the x/y values, for example the median of the
     x values (0.50 quantile) and some other quantile of the y values. The
     default (recommended) behavior is to keep the x-quantile = 0.5 and vary
     the y-quantile to move the line up and down as desired to define an
     "envelope"

 Example

  Generate test data with known parameters a and b, then fit them:

     [t, q, dqdt] = baseflow.generateTestData(1e-2, 1.5, 100);
     Fit = baseflow.fitab(q, dqdt, 'nls');
     fprintf('a = %.4f, b = %.2f\n', Fit.a, Fit.b)

0149 function [ab,ci,ok] = fitOLS(logx,logy,weights,alpha)
0150    % ordinary least squares linear regression in log-log
0151    %
0152    % Solve weighted least squares with plain linear algebra, so the method
0153    % runs on MATLAB and Octave. The result matches the Curve Fitting Toolbox
0154    % confint(fit(logx,logy,'poly1'), alpha) with fitoptions Weights.
0155 
0156    % Scale each row by the square root of its weight, so the unweighted
0157    % solve minimizes sum(weights.*resid.^2). Masked points have zero weight
0158    % and add nothing to the coefficients or the residuals.
0159    sw = sqrt(weights);
0160    X = [sw, sw.*logx];
0161    [Qx,R] = qr(X,0);
0162    ab = R \ (Qx'*(sw.*logy));
0163    resid = sw.*logy - X*ab;
0164 
0165    % Compute coefficient standard errors from inv(X'*X) = inv(R)*inv(R)'.
0166    % fit counts zero-weight (masked) points in the n-2 residual degrees of
0167    % freedom, so count them here too to match confint.
0168    dfe = numel(logy) - 2;
0169    Rinv = R \ eye(2);
0170    se = sqrt(diag(Rinv*Rinv') * sum(resid.^2) / dfe);
0171 
0172    % Compute t-based intervals at confidence level alpha, one row per
0173    % coefficient, in the order [intercept; slope]. This is the rot90(confint)
0174    % layout, which is consistent with the stats functions.
0175    tcrit = tinv((1 + alpha) / 2, dfe);
0176    ci = [ab - tcrit*se, ab + tcrit*se];
0177 
0178    % transform a to linear space and package a/b
0179    ab = [exp(ab(1)); ab(2)];
0180    ci(1,:) = exp(ci(1,:));
0181 
0182    % generic failure check
0183    ok = all(isreal(ab));
0184 end

0484       ab1ok = false;
0485       % rethrow(ME)
0486    end
0487 
0488    % if nlinfit worked and it's 'better' than rsq (here, rsq0), select it
0489    if ab1ok && rsq1 > rsq && rsq1 > 0
0490 
0491       fselect = 'nlinfit_robust';
0492       rsq = rsq1;
0493    else
0494 
0495       % try curve fitting functions
0496       ab3ok = true;
0497       try
0498          f3 = fit(x,y,ftype,opts3);
0499          ab3 = coeffvalues(f3);
0500          rsq3 = baseflow.deps.rsquare(y,ab3(1).*x.^ab3(2));
0501 
0502       catch ME
0503 
0504          if (strcmp(ME.identifier,'curvefit:fit:infComputed'))
0505 
0506             msg = 'Fitting failed using fit at ab3';
0507             causeException = MException('BFRA:fitab:fitting',msg);
0508             ME = addCause(ME,causeException);
0509          end
0510          ab3ok = false;
0511          % rethrow(ME)
0512       end
0513 
0514       % if fit worked, select it
0515       if ab3ok && rsq3 > rsq && rsq3 > 0
0516 
0517          fselect = 'fit_robust';
0518          rsq = rsq3;
0519       else
0520          fselect = 'none';
0521 
0522       end
0523    end
0524 
0525    % if neither nlinfit nor fit worked, try non-robust fitting
0526    if strcmp(fselect,'none')
0527 
0528       ab2ok = true;
0529       try
0530          [ab2,R2,~,C2] = nlinfit(x,y,fnc,ab0,opts2);
0531          rsq2 = baseflow.deps.rsquare(y,ab2(1).*x.^ab2(2));
0532 
0533       catch ME
0534 
0535          if (strcmp(ME.identifier,'stats:nlinfit:NoUsableObservations'))
0536 
0537             msg = 'Fitting failed using nlinfit at ab2';
0538             causeException = MException('BFRA:fitab:fitting',msg);
0539             ME = addCause(ME,causeException);
0540          end
0541          ab2ok = false;
0542          %rethrow(ME)
0543       end
0544 
0545       if ab2ok && rsq2 > rsq && rsq2 > 0
0546 
0547          fselect = 'nlinfit';
0548          rsq = rsq2;
0549 
0550       else % try curve fitting functions
0551          ab4ok = true;
0552          try
0553             f4 = fit(x,y,ftype,opts4); ab4 = coeffvalues(f4);
0554             rsq4 = baseflow.deps.rsquare(y,ab4(1).*x.^ab4(2));
0555 
0556          catch ME
0557             if (strcmp(ME.identifier,'curvefit:fit:infComputed'))
0558 
0559                msg = 'Fitting failed using fit at ab4';
0560                causeException = MException('BFRA:fitab:fitting',msg);
0561                ME = addCause(ME,causeException);
0562             end
0563             ab4ok = false;
0564             %rethrow(ME)
0565          end
0566 
0567          % we don't compare with rsq2 because we already know its <rsq0
0568          if ab4ok && rsq4 > rsq && rsq4 > 0
0569             fselect = 'fit';
0570             rsq = rsq4;
0571          else
0572             fselect = 'none';
0573          end
0574       end
0575    end
0576 
0577    % finally, if rsq is still low but linear rsq is good, choose lin
0578    if strcmp(fselect,'none') && rsq > 0
0579       fselect = 'linear';
0580    elseif rsq < 0
0581       % NOTE: nov 2022, i think in some cases we can get here and rsq < 0 so I
0582       % added this option , previously there was no else, just end
0583       fselect = 'none';
0584    end
0585 
0586    switch fselect
0587 
0588       case 'none'
0589          ok = false;   % should never occur with option linear
0590          ab = [nan,nan]; % turns out can occur if q vs dqdt is decreasing
0591          ci = [nan nan; nan nan];
0592 
0593       case 'nlinfit_robust'
0594          ci = nlparci(ab1,R1,'covariance',C1,'alpha',alpha);
0595          ab = ab1;
0596 
0597       case 'nlinfit'
0598          ci = nlparci(ab2,R2,'covariance',C2,'alpha',0.68);
0599          ab = ab2;
0600 
0601       case 'fit_robust'
0602          ab = ab3;
0603          ci = transpose(confint(f3,alpha));
0604 
0605       case 'fit'
0606          ab = ab4;
0607          ci = transpose(confint(f4,alpha));
0608 
0609       case 'linear'
0610 
0611          [ab,ci] = regress(logy,[ones(size(y)) logx]);
0612          ci(1,:) = exp(ci(1,:));
0613          ab = [exp(ab(1)); ab(2)];
0614 
0615          % might check metrics such as islineconvex(y), and if x<xmin where
0616          % xmin is some very small flow value below which the data is corrupt
0617    end

0697 function ci = nlparci_octave(beta, CovB, alpha)
0698 
0699    % INPUTS:
0700    %   beta: coefficients from nlinfit
0701    %   CovB: covariance matrix from nlinfit
0702    %   alpha: desired confidence level (e.g., 0.68 for 68%)
0703    %
0704    % OUTPUTS:
0705    %   ci_lower: lower bounds of the confidence intervals
0706    %   ci_upper: upper bounds of the confidence intervals
0707 
0708    n = length(beta); % Number of coefficients
0709    dof = n - 1; % Degrees of freedom
0710    % t-score for desired confidence level
0711    t_score = tinv(1 - (1 - alpha) / 2, dof);
0712    se = sqrt(diag(CovB)); % Standard errors of the coefficients
0713    ci_lower = beta' - t_score * se; % Lower bounds of the confidence intervals
0714    ci_upper = beta' + t_score * se; % Upper bounds of the confidence intervals
0715    ci = [ci_lower, ci_upper];
0716 end
